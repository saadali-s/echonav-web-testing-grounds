package datastructures;

public class AdjacencyListGraph<T> implements IGraph<T> {

    // We use your custom ArrayList to hold all the nodes in the map!
    private ArrayList<INode<T>> nodes;

    public AdjacencyListGraph() {
        this.nodes = new ArrayList<INode<T>>();
    }

    @Override
    public INode<T> addNode(T data) {
        GraphNode<T> newNode = new GraphNode<>(data);
        // Add to the end of our list
        nodes.add(nodes.size(), newNode);
        return newNode;
    }

    @Override
    public void addEdge(INode<T> source, INode<T> destination, double weight) {
        // We cast to GraphNode because we need access to that secret 'edges' list we made
        GraphNode<T> srcNode = (GraphNode<T>) source;
        
        // Create the new connection and add it to the source's personal list
        Edge<T> newEdge = new Edge<>(destination, weight);
        srcNode.getEdges().add(srcNode.getEdges().size(), newEdge);
        
        // Note: This is a DIRECTED edge. If we want an UNDIRECTED graph (like a road where 
        // you can drive both ways), we would add the reverse edge here too. 
        // For RuralRoute SA, let's make it undirected so the ambulance can drive back!
        GraphNode<T> destNode = (GraphNode<T>) destination;
        Edge<T> reverseEdge = new Edge<>(source, weight);
        destNode.getEdges().add(destNode.getEdges().size(), reverseEdge);
    }

    @Override
    public List<INode<T>> getNodes() {
        return nodes;
    }

    @Override
    public List<INode<T>> getNeighbors(INode<T> node) {
        GraphNode<T> gNode = (GraphNode<T>) node;
        ArrayList<INode<T>> neighbors = new ArrayList<>();
        
        // Loop through the edges and pull out just the destination nodes
        for (int i = 0; i < gNode.getEdges().size(); i++) {
            neighbors.add(neighbors.size(), gNode.getEdges().get(i).getDestination());
        }
        return neighbors;
    }

    @Override
    public double getEdgeWeight(INode<T> source, INode<T> destination) {
        GraphNode<T> srcNode = (GraphNode<T>) source;
        
        // Search through the source's edges to find the one pointing to the destination
        for (int i = 0; i < srcNode.getEdges().size(); i++) {
            Edge<T> edge = srcNode.getEdges().get(i);
            if (edge.getDestination().equals(destination)) {
                return edge.getWeight();
            }
        }
        // If no edge exists, the cost to travel there is infinity!
        return Double.MAX_VALUE; 
    }

    @Override
    public boolean removeNode(INode<T> node) {
        // Removing a node from an Adjacency List is complex because you have to 
        // find and delete all edges pointing TO it first. 
        // For our RuralRoute map, we actually never delete land blocks, we just 
        // change their weight to infinity. So we can leave this unimplemented for now!
        throw new UnsupportedOperationException("Nodes are permanent on the map!");
    }

    @Override
    public boolean removeEdge(INode<T> source, INode<T> destination) {
        throw new UnsupportedOperationException("Edges are permanent on the map!");
    }

package datastructures;

import java.util.Iterator;

public class ArrayList<T> implements List<T>, Iterable<T> {

	private T[] array;
	private Integer size;
	private Integer arrayLength;
	private Integer strategy;
	
	public ArrayList() {
		this(1);
	}
	
	public ArrayList(Integer strategy) {
		this.strategy = strategy;
		this.arrayLength = 1;
		this.array = createArray(this.arrayLength);
		this.size = 0;
	}
	
	@SuppressWarnings("unchecked")
	private T[] createArray(int size) {
		return (T[]) new Object[size];
	}
	
	@Override
	public T get(Integer i) {
		if (i >= size) {
			throw new ArrayListException("Index greater than size");
		}
		if (i < 0) {
			throw new ArrayListException("Index out of range");
		}
		return array[i];
	}

	@Override
	public void set(Integer i, T e) {
		if (i >= size) {
			throw new ArrayListException("Index greater than size");
		}
		if (i < 0) {
			throw new ArrayListException("Index out of range");
		}
		array[i] = e;
	}

	@Override
	public void add(Integer i, T e) {
		if (i < 0 || i > size) {
			throw new ArrayListException("Index out of range");
		}
		if (size >= arrayLength) {
			expandArray();
		}
		shiftElementsRight(i);
		array[i] = e;
		size++;
	}

	@Override
	public T remove(Integer i) {
		if (i < 0 || i >= size) {
			throw new ArrayListException("Index out of range");
		}
		
		T elementToRemove = array[i];
		shiftElementsLeft(i);
		size--;
		
		return elementToRemove;
	}

	@Override
	public Integer size() {
		return size;
	}

	@Override
	public boolean isEmpty() {
		return size == 0;
	}
	
	public String toString() {
		String str = "[";
		for (int i = 0; i < size-1; i++) {
			str += array[i].toString() + ",";
		}
		if (size > 0) {
			str += array[size-1];
		}
		str += "]";
		return str;
	}
	
	private void expandArray() {
		if (strategy == 1) {
			this.arrayLength += 10; 
		} else if (strategy == 2) {
			this.arrayLength *= 2;
		}
		
		T[] newArray = createArray(this.arrayLength);
		System.arraycopy(this.array, 0, newArray, 0, this.size); 
		this.array = newArray;
	}
	
	private void shiftElementsRight(Integer pos) {
		for (int i = this.size - 1; i >= pos; i--) {
			this.array[i+1] = this.array[i];
		}
	}
	
	private void shiftElementsLeft(Integer pos) {
		for (int i = pos; i < size - 1; i++) {
			this.array[i] = this.array[i+1];
		}
		if (size > 0) {
			this.array[size - 1] = null; 
		}
	}

	@Override
	public Iterator<T> iterator() {		
		return new ArrayListIterator<T>(this);
	}
}.

package datastructures;

import java.util.Iterator;
import java.util.NoSuchElementException;

public class ArrayListIterator<T> implements Iterator<T> {
	private ArrayList<T> list;
	private int cursor;
	
	public ArrayListIterator(ArrayList<T> list) {
		this.list = list;
		if (!list.isEmpty()) {
			this.cursor = 0;
		}
	}
	
	@Override
	public boolean hasNext() {
		return cursor < list.size();
	}

	@Override
	public T next() {
		if (!hasNext()) {
			throw new NoSuchElementException("No more elements in the list");
		}
		T item = list.get(cursor);
		cursor++;
		return item;
	}

	@Override
	public void remove() {
		//do nothing - keep java happy
	}
}.

package datastructures;

import java.util.Arrays;

public class Matrix implements IMatrix {
    
    private double[][] data;
    private int rows;
    private int cols;
    
    public Matrix() {
        this.rows = 0;
        this.cols = 0;
        this.data = new double[0][0];
    }
    
    public Matrix(int rows, int cols) {
        this.rows = rows;
        this.cols = cols;
        this.data = new double[rows][cols];
    }
    
    @Override
    public int numberRows() { return rows; }
    
    @Override
    public int numberCols() { return cols; }
    
    @Override
    public IMatrix getRow(int i) throws MatrixException {
        if (i < 0 || i >= rows) throw new MatrixException("Invalid row index");
        Matrix result = new Matrix(1, cols);
        for (int j = 0; j < cols; j++) result.data[0][j] = data[i][j];
        return result;
    }
    
    @Override
    public IMatrix getCol(int j) throws MatrixException {
        if (j < 0 || j >= cols) throw new MatrixException("Invalid column index");
        Matrix result = new Matrix(rows, 1);
        for (int i = 0; i < rows; i++) result.data[i][0] = data[i][j];
        return result;
    }
    
    @Override
    public double getElement(int i, int j) throws MatrixException {
        if (i < 0 || i >= rows || j < 0 || j >= cols) throw new MatrixException("Index out of bounds");
        return data[i][j];
    }
    
    public void setElement(int i, int j, double value) throws MatrixException {
        if (i < 0 || i >= rows || j < 0 || j >= cols) throw new MatrixException("Index out of bounds");
        data[i][j] = value;
    }
    
    @Override
    public IMatrix addMatrix(IMatrix otherMatrix) throws MatrixException {
        if (rows != otherMatrix.numberRows() || cols != otherMatrix.numberCols()) throw new MatrixException("Dimension mismatch");
        Matrix result = new Matrix(rows, cols);
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) result.data[i][j] = data[i][j] + otherMatrix.getElement(i, j);
        }
        return result;
    }
    
    @Override
    public IMatrix addScalar(double c) {
        Matrix result = new Matrix(rows, cols);
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) result.data[i][j] = data[i][j] + c;
        }
        return result;
    }
    
    @Override
    public IMatrix multiplyScalar(double c) {
        Matrix result = new Matrix(rows, cols);
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) result.data[i][j] = data[i][j] * c;
        }
        return result;
    }
    
    @Override
    public IMatrix multiplyMatrix(IMatrix otherMatrix) throws MatrixException {
        if (cols != otherMatrix.numberRows()) throw new MatrixException("Dimension mismatch");
        Matrix result = new Matrix(rows, otherMatrix.numberCols());
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < otherMatrix.numberCols(); j++) {
                double sum = 0;
                for (int k = 0; k < cols; k++) sum += data[i][k] * otherMatrix.getElement(k, j);
                result.data[i][j] = sum;
            }
        }
        return result;
    }
    
    @Override
    public IMatrix transpose() {
        Matrix result = new Matrix(cols, rows);
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) result.data[j][i] = data[i][j];
        }
        return result;
    }

package datastructures;

public class MinHeap<T> {
    
    // An internal class to hold a node and its current distance priority
    public static class HeapElement<T> {
        public INode<T> node;
        public double priority;

        public HeapElement(INode<T> node, double priority) {
            this.node = node;
            this.priority = priority;
        }
    }

    private ArrayList<HeapElement<T>> heap;

    public MinHeap() {
        heap = new ArrayList<>();
    }

    public void insert(INode<T> node, double priority) {
        HeapElement<T> newElement = new HeapElement<>(node, priority);
        heap.add(heap.size(), newElement);
        bubbleUp(heap.size() - 1);
    }

    public HeapElement<T> extractMin() {
        if (heap.isEmpty()) return null;

        HeapElement<T> min = heap.get(0);
        HeapElement<T> last = heap.remove(heap.size() - 1);

        if (!heap.isEmpty()) {
            heap.set(0, last);
            bubbleDown(0);
        }
        return min;
    }

    public boolean isEmpty() {
        return heap.isEmpty();
    }

    private void bubbleUp(int index) {
        int parent = (index - 1) / 2;
        while (index > 0 && heap.get(index).priority < heap.get(parent).priority) {
            swap(index, parent);
            index = parent;
            parent = (index - 1) / 2;
        }
    }

    private void bubbleDown(int index) {
        int size = heap.size();
        while (true) {
            int leftChild = 2 * index + 1;
            int rightChild = 2 * index + 2;
            int smallest = index;

            if (leftChild < size && heap.get(leftChild).priority < heap.get(smallest).priority) {
                smallest = leftChild;
            }
            if (rightChild < size && heap.get(rightChild).priority < heap.get(smallest).priority) {
                smallest = rightChild;
            }

            if (smallest != index) {
                swap(index, smallest);
                index = smallest;
            } else {
                break;
            }
        }
    }

    private void swap(int i, int j) {
        HeapElement<T> temp = heap.get(i);
        heap.set(i, heap.get(j));
        heap.set(j, temp);
    }
}.

package imageprocessing;

import datastructures.AdjacencyListGraph;
import datastructures.INode;
import datastructures.Matrix;
import datastructures.MatrixException;

public class GraphBuilder {

    public static AdjacencyListGraph<RegionData> buildVascularGraph(
            Matrix greenChannel, Matrix blockBinaryMask, int blockSize) {
        
        AdjacencyListGraph<RegionData> graph = new AdjacencyListGraph<>();
        int gridRows = greenChannel.numberRows() / blockSize;
        int gridCols = greenChannel.numberCols() / blockSize;
        
        @SuppressWarnings("unchecked")
        INode<RegionData>[][] nodeGrid = new INode[gridRows][gridCols];

        // PASS 1: Build nodes directly from the Block-Scaled Skeleton
        for (int r = 0; r < gridRows; r++) {
            for (int c = 0; c < gridCols; c++) {
                try {
                    if (blockBinaryMask.getElement(r, c) == 1.0) {
                        // Grab an approximate green intensity from the center of the block
                        int centerY = Math.min(r * blockSize + (blockSize / 2), greenChannel.numberRows() - 1);
                        int centerX = Math.min(c * blockSize + (blockSize / 2), greenChannel.numberCols() - 1);
                        double minGreen = greenChannel.getElement(centerY, centerX);
                        
                        RegionData data = new RegionData(c, r, minGreen);
                        data.setVessel(true); 
                        nodeGrid[r][c] = graph.addNode(data);
                    }
                } catch (MatrixException e) {}
            }
        }

        // PASS 2: 8-Way Networking
        for (int r = 0; r < gridRows; r++) {
            for (int c = 0; c < gridCols; c++) {
                INode<RegionData> currentNode = nodeGrid[r][c];
                if (currentNode == null) continue;

                for (int dr = -1; dr <= 1; dr++) {
                    for (int dc = -1; dc <= 1; dc++) {
                        if (dr == 0 && dc == 0) continue; 

                        int nr = r + dr;
                        int nc = c + dc;

                        if (nr >= 0 && nr < gridRows && nc >= 0 && nc < gridCols) {
                            if (nodeGrid[nr][nc] != null) {
                                double weight = (dr == 0 || dc == 0) ? 1.0 : 1.414;
                                graph.addEdge(currentNode, nodeGrid[nr][nc], weight);
                            }
                        }
                    }
                }
            }
        }

        // PASS 3: GRAPH ENDPOINT BRIDGING
        datastructures.List<INode<RegionData>> allNodes = graph.getNodes();
        ArrayList<INode<RegionData>> endpoints = new ArrayList<>();
        
        for (int i = 0; i < allNodes.size(); i++) {
            if (graph.getNeighbors(allNodes.get(i)).size() == 1) {
                endpoints.add(endpoints.size(), allNodes.get(i));
            }
        }

        double bridgeThreshold = 4.0; 

        for (int i = 0; i < endpoints.size(); i++) {
            INode<RegionData> endA = endpoints.get(i);
            RegionData dataA = endA.getData();
            
            for (int j = i + 1; j < endpoints.size(); j++) {
                INode<RegionData> endB = endpoints.get(j);
                RegionData dataB = endB.getData();
                
                double dist = Math.sqrt(Math.pow(dataA.getGridX() - dataB.getGridX(), 2) 
                                      + Math.pow(dataA.getGridY() - dataB.getGridY(), 2));
                
                if (dist <= bridgeThreshold) {
                    graph.addEdge(endA, endB, dist);
                }
            }
        }

        return graph;
    }

    public static void mergeDiagnosticData(
            AdjacencyListGraph<RegionData> graph, 
            INode<RegionData>[][] nodeGrid, 
            ArrayList<RegionData> anomalies) {
        
        for (int i = 0; i < anomalies.size(); i++) {
            RegionData lesion = anomalies.get(i);
            int r = lesion.getGridY();
            int c = lesion.getGridX();

            if (nodeGrid[r][c] != null) {
                nodeGrid[r][c].getData().setAnomaly(true);
            } else {
                INode<RegionData> newNode = graph.addNode(lesion);
                nodeGrid[r][c] = newNode;
            }
        }
    }

    public static Matrix applySauvolaThresholding(Matrix greenChannel) throws MatrixException {
        int height = greenChannel.numberRows();
        int width = greenChannel.numberCols();
        Matrix binaryMatrix = new Matrix(height, width);

        int W = 45; 
        double k = 0.12; 
        double R = 128.0; 

        double[][] intSum = new double[height][width];
        double[][] intSqSum = new double[height][width];

        for (int r = 0; r < height; r++) {
            for (int c = 0; c < width; c++) {
                double val = greenChannel.getElement(r, c);
                
                double topSum = (r > 0) ? intSum[r - 1][c] : 0;
                double leftSum = (c > 0) ? intSum[r][c - 1] : 0;
                double topLeftSum = (r > 0 && c > 0) ? intSum[r - 1][c - 1] : 0;
                intSum[r][c] = val + topSum + leftSum - topLeftSum;

                double topSqSum = (r > 0) ? intSqSum[r - 1][c] : 0;
                double leftSqSum = (c > 0) ? intSqSum[r][c - 1] : 0;
                double topLeftSqSum = (r > 0 && c > 0) ? intSqSum[r - 1][c - 1] : 0;
                intSqSum[r][c] = (val * val) + topSqSum + leftSqSum - topLeftSqSum;
            }
        }

        int offset = W / 2;
        for (int r = 0; r < height; r++) {
            for (int c = 0; c < width; c++) {
                double currentVal = greenChannel.getElement(r, c);
                
                if (currentVal < 20.0) {
                    binaryMatrix.setElement(r, c, 0.0);
                    continue;
                }

                int r1 = Math.max(0, r - offset);
                int c1 = Math.max(0, c - offset);
                int r2 = Math.min(height - 1, r + offset);
                int c2 = Math.min(width - 1, c + offset);

                double area = (r2 - r1 + 1) * (c2 - c1 + 1);

                double topSum = (r1 > 0) ? intSum[r1 - 1][c2] : 0;
                double leftSum = (c1 > 0) ? intSum[r2][c1 - 1] : 0;
                double topLeftSum = (r1 > 0 && c1 > 0) ? intSum[r1 - 1][c1 - 1] : 0;
                double sum = intSum[r2][c2] - topSum - leftSum + topLeftSum;

                double topSqSum = (r1 > 0) ? intSqSum[r1 - 1][c2] : 0;
                double leftSqSum = (c1 > 0) ? intSqSum[r2][c1 - 1] : 0;
                double topLeftSqSum = (r1 > 0 && c1 > 0) ? intSqSum[r1 - 1][c1 - 1] : 0;
                double sqSum = intSqSum[r2][c2] - topSqSum - leftSqSum + topLeftSqSum;

                double mean = sum / area;
                double variance = (sqSum / area) - (mean * mean);
                double stdDev = Math.sqrt(Math.max(0, variance));

                double threshold = mean * (1.0 - k * (1.0 - (stdDev / R)));

                if (currentVal < threshold) {
                    binaryMatrix.setElement(r, c, 1.0); 
                } else {
                    binaryMatrix.setElement(r, c, 0.0); 
                }
            }
        }
        return binaryMatrix;
    }

import datastructures.IGraph;

public class PathFinder {

    // =========================================================================
    // 1. DIJKSTRA'S ALGORITHM
    // =========================================================================
    public static List<INode<RegionData>> calculateTortuosity(
            IGraph<RegionData> graph, INode<RegionData> startNode, INode<RegionData> endNode, 
            int gridRows, int gridCols) {
        
        double[][] distances = new double[gridRows][gridCols];
        @SuppressWarnings("unchecked")
        INode<RegionData>[][] previousNode = new INode[gridRows][gridCols];
        
        for (int r = 0; r < gridRows; r++) {
            for (int c = 0; c < gridCols; c++) {
                distances[r][c] = Double.MAX_VALUE;
                previousNode[r][c] = null;
            }
        }

        MinHeap<RegionData> pq = new MinHeap<>();
        RegionData startData = startNode.getData();
        distances[startData.getGridY()][startData.getGridX()] = 0.0;
        pq.insert(startNode, 0.0);

        while (!pq.isEmpty()) {
            MinHeap.HeapElement<RegionData> currentElement = pq.extractMin();
            INode<RegionData> currentNode = currentElement.node;
            RegionData currentData = currentNode.getData();
            
            int curX = currentData.getGridX();
            int curY = currentData.getGridY();

            if (currentNode.equals(endNode)) break;
            if (currentElement.priority > distances[curY][curX]) continue;

            List<INode<RegionData>> neighbors = graph.getNeighbors(currentNode);
            for (int i = 0; i < neighbors.size(); i++) {
                INode<RegionData> neighbor = neighbors.get(i);
                RegionData neighborData = neighbor.getData();
                
                if (!neighborData.isVessel()) continue; 

                int nX = neighborData.getGridX();
                int nY = neighborData.getGridY();

                double edgeWeight = graph.getEdgeWeight(currentNode, neighbor);
                double newPathDistance = distances[curY][curX] + edgeWeight;

                if (newPathDistance < distances[nY][nX]) {
                    distances[nY][nX] = newPathDistance;
                    previousNode[nY][nX] = currentNode;
                    pq.insert(neighbor, newPathDistance);
                }
            }
        }

        ArrayList<INode<RegionData>> finalPath = new ArrayList<>();
        INode<RegionData> step = endNode;
        
        while (step != null) {
            finalPath.add(0, step); 
            if (step.equals(startNode)) break; 
            RegionData stepData = step.getData();
            step = previousNode[stepData.getGridY()][stepData.getGridX()];
        }

        if (finalPath.size() <= 1 && !startNode.equals(endNode)) {
            return new ArrayList<>(); 
        }
        return finalPath;
    }

    // 2. EXHAUSTIVE TRACER
    // =========================================================================
    public static double calculateGlobalChaos(IGraph<RegionData> graph, int gridRows, int gridCols, ArrayList<List<INode<RegionData>>> longestPathOut) {
        List<INode<RegionData>> nodes = graph.getNodes();
        if (nodes.size() < 2) return 1.0;

        double totalActual = 0.0;
        double totalEuclid = 0.0;
        int validPaths = 0;

        ArrayList<INode<RegionData>> endpoints = new ArrayList<>();
        for (int i = 0; i < nodes.size(); i++) {
            INode<RegionData> n = nodes.get(i);
            if (graph.getNeighbors(n).size() == 1) {
                endpoints.add(endpoints.size(), n);
            }
        }

        List<INode<RegionData>> longestPath = null;
        ArrayList<INode<RegionData>> processedEnds = new ArrayList<>();

        for (int i = 0; i < endpoints.size(); i++) {
            INode<RegionData> startNode = endpoints.get(i);
            if (listContains(processedEnds, startNode)) continue;

            INode<RegionData> furthestNode = findFurthestNode(graph, startNode);
            
            if (furthestNode != null && !furthestNode.equals(startNode)) {
                processedEnds.add(processedEnds.size(), startNode);
                processedEnds.add(processedEnds.size(), furthestNode);

                RegionData d1 = startNode.getData();
                RegionData d2 = furthestNode.getData();
                double euclidLength = Math.sqrt(Math.pow(d1.getGridX() - d2.getGridX(), 2) 
                                              + Math.pow(d1.getGridY() - d2.getGridY(), 2));

                // 10 blocks = 40 pixels. Perfect filter size for finding main vessels.
                if (euclidLength > 10.0) {
                    List<INode<RegionData>> dijkstraPath = calculateTortuosity(graph, startNode, furthestNode, gridRows, gridCols);
                    
                    if (dijkstraPath.size() > 0) {
                        
                        double actualPathLength = 0.0;
                        for (int k = 0; k < dijkstraPath.size() - 1; k++) {
                            RegionData n1 = dijkstraPath.get(k).getData();
                            RegionData n2 = dijkstraPath.get(k+1).getData();
                            actualPathLength += Math.sqrt(Math.pow(n1.getGridX() - n2.getGridX(), 2) 
                                                        + Math.pow(n1.getGridY() - n2.getGridY(), 2));
                        }

                        actualPathLength = Math.max(actualPathLength, euclidLength);

                        totalActual += actualPathLength;
                        totalEuclid += euclidLength;
                        validPaths++;
                        
                        if (longestPath == null || dijkstraPath.size() > longestPath.size()) {
                            longestPath = dijkstraPath;
                        }
                    }
                }
            }
        }

        if (longestPath != null && longestPathOut != null) {
            longestPathOut.add(0, longestPath);
        }

        if (validPaths == 0 || totalEuclid == 0) return 1.0;
        return totalActual / totalEuclid;
    }

    // =========================================================================
    // 3. NEOVASCULARIZATION DETECTOR (Euler Characteristic)
    // =========================================================================
    public static int countNeovascularLoops(IGraph<RegionData> graph, int gridRows, int gridCols) {
        List<INode<RegionData>> nodes = graph.getNodes();
        if (nodes.size() == 0) return 0;

        int V = nodes.size();
        int E_total = 0;
        
        for (int i = 0; i < nodes.size(); i++) {
            E_total += graph.getNeighbors(nodes.get(i)).size();
        }
        int E = E_total / 2;

        int C = 0;
        boolean[][] visited = new boolean[gridRows][gridCols];
        
        for (int i = 0; i < nodes.size(); i++) {
            RegionData d = nodes.get(i).getData();
            if (!visited[d.getGridY()][d.getGridX()]) {
                C++;
                ArrayList<INode<RegionData>> queue = new ArrayList<>();
                queue.add(queue.size(), nodes.get(i));
                visited[d.getGridY()][d.getGridX()] = true;
                
                while (!queue.isEmpty()) {
                    INode<RegionData> curr = queue.remove(0);
                    List<INode<RegionData>> neighbors = graph.getNeighbors(curr);
                    for (int j = 0; j < neighbors.size(); j++) {
                        INode<RegionData> neighbor = neighbors.get(j);
                        RegionData nd = neighbor.getData();
                        if (!visited[nd.getGridY()][nd.getGridX()]) {
                            visited[nd.getGridY()][nd.getGridX()] = true;
                            queue.add(queue.size(), neighbor);
                        }
                    }
                }
            }
        }

        int cycles = E - V + C;
        return Math.max(0, cycles);
    }

    private static INode<RegionData> findFurthestNode(IGraph<RegionData> graph, INode<RegionData> start) {
        ArrayList<INode<RegionData>> queue = new ArrayList<>();
        ArrayList<INode<RegionData>> visited = new ArrayList<>();
        
        queue.add(queue.size(), start);
        visited.add(visited.size(), start);
        INode<RegionData> furthest = start;

        while (!queue.isEmpty()) {
            INode<RegionData> current = queue.remove(0);
            furthest = current;
            
            List<INode<RegionData>> neighbors = graph.getNeighbors(current);
            for (int i = 0; i < neighbors.size(); i++) {
                INode<RegionData> n = neighbors.get(i);
                if (!listContains(visited, n)) {
                    visited.add(visited.size(), n);
                    queue.add(queue.size(), n);
                }
            }
        }
        return furthest;
    }

    private static boolean listContains(ArrayList<INode<RegionData>> list, INode<RegionData> node) {
        for (int i = 0; i < list.size(); i++) {
            if (list.get(i).equals(node)) return true;
        }
        return false;
    }

import datastructures.ArrayList;

import datastructures.MatrixException;

public class SimilarityDetector {

    // Define the rigid constant for k in the k-NN clustering logic
    private static final int K_NEIGHBORS = 3;

    /**
     * Internal helper methodology to architect a synthetic diagnostic prototype node.
     * The gridX integer is co-opted as a formal class label: 
     * 0 = Healthy Tissue, 1 = Red Lesion Pathology, 2 = Bright Lesion Pathology
     * The F1 (Green) metric is stored in averageBrightness.
     * The F2 (Red) metric is stored in gradientIntensity.
     */
    private static INode<RegionData> createPrototype(AdjacencyListGraph<RegionData> graph, 
                                                     int classLabel, double f1Green, double f2Red, boolean isAnomaly) {
        RegionData protoData = new RegionData(classLabel, -1, f1Green);
        protoData.setGradientIntensity(f2Red);
        protoData.setAnomaly(isAnomaly);
        return graph.addNode(protoData);
    }

    public static ArrayList<RegionData> scanForAnomalies(Matrix greenChannel, Matrix redChannel, int blockSize) {
        ArrayList<RegionData> anomalies = new ArrayList<>();
        
        int gridRows = greenChannel.numberRows() / blockSize;
        int gridCols = greenChannel.numberCols() / blockSize;
        
        // PASS 1: Calculate localized block averages to establish the F1 and F2 base data layers
        double[][] greenBlocks = new double[gridRows][gridCols];
        double[][] redBlocks = new double[gridRows][gridCols];
        
        for (int r = 0; r < gridRows; r++) {
            for (int c = 0; c < gridCols; c++) {
                double gSum = 0, rSum = 0;
                int count = 0;
                for (int y = r * blockSize; y < (r + 1) * blockSize; y++) {
                    for (int x = c * blockSize; x < (c + 1) * blockSize; x++) {
                        try {
                            gSum += greenChannel.getElement(y, x);
                            rSum += redChannel.getElement(y, x);
                            count++;
                        } catch (MatrixException e) { 
                            // Boundary safely caught
                        }
                    }
                }
                greenBlocks[r][c] = gSum / count;
                redBlocks[r][c] = rSum / count;
            }
        }

        // --- NEW PASS 1.5: Dynamic Baseline Calibration ---
        double globalGreenSum = 0, globalRedSum = 0;
        int validBlocks = 0;
        int radius = 2; 
        
        for (int r = radius; r < gridRows - radius; r++) {
            for (int c = radius; c < gridCols - radius; c++) {
                // Ignore void (black) and optic disc (extreme bright) when calculating the baseline
                if (greenBlocks[r][c] > 20.0 && !(redBlocks[r][c] > 220.0 && greenBlocks[r][c] > 200.0)) {
                    globalGreenSum += greenBlocks[r][c];
                    globalRedSum += redBlocks[r][c];
                    validBlocks++;
                }
            }
        }
        
        // Calculate this specific image's normal lighting
        double baseG = validBlocks > 0 ? globalGreenSum / validBlocks : 90.0;
        double baseR = validBlocks > 0 ? globalRedSum / validBlocks : 150.0;

        // PASS 2: Similarity Graph Initialization & DYNAMIC Prototype Definition
        AdjacencyListGraph<RegionData> knnGraph = new AdjacencyListGraph<>();
        ArrayList<INode<RegionData>> prototypes = new ArrayList<>();
        
        // State array to store the isolated F3 contrast values for the 6 theoretical prototypes
        double[] protoF3 = new double[6]; 

        // Establish Prototype 0, 1: Healthy Retina Tissue (Based on actual image lighting)
        prototypes.add(prototypes.size(), createPrototype(knnGraph, 0, baseG, baseR, false));
        protoF3[0] = 0.0;
        prototypes.add(prototypes.size(), createPrototype(knnGraph, 0, baseG + 10.0, baseR + 10.0, false));
        protoF3[1] = 0.0;

        // Establish Prototype 2, 3: Red Lesions (Significantly darker than the baseline)
        prototypes.add(prototypes.size(), createPrototype(knnGraph, 1, baseG - 40.0, baseR - 40.0, true));
        protoF3[2] = +35.0;
        prototypes.add(prototypes.size(), createPrototype(knnGraph, 1, baseG - 60.0, baseR - 60.0, true));
        protoF3[3] = +50.0;

        // Establish Prototype 4, 5: Bright Lesions (Significantly brighter than the baseline)
        prototypes.add(prototypes.size(), createPrototype(knnGraph, 2, baseG + 60.0, baseR + 60.0, true));
        protoF3[4] = -45.0;

                }

                while (!pq.isEmpty()) {

                    }
                    
                    anomalies.add(anomalies.size(), lesion);
                }
            }
        }
        
        System.out.println("k-NN Similarity Graph Generation Complete: Extracted " + anomalies.size() + " pathological lesions.");
