import { useCallback, useState } from "react";
import { EdgeData, NodeData } from "../vector/types";

export function useHistory(initialNodes: NodeData[], initialEdges: EdgeData[]) {
  const [history, setHistory] = useState<{ nodes: NodeData[]; edges: EdgeData[] }[]>([
    { nodes: initialNodes, edges: initialEdges }
  ]);
  const [historyIndex, setHistoryIndex] = useState(0);

  const pushHistory = useCallback((newNodes: NodeData[], newEdges: EdgeData[]) => {
    setHistory((prev) => {
      const newHistory = prev.slice(0, historyIndex + 1);
      newHistory.push({ nodes: newNodes, edges: newEdges });
      return newHistory;
    });
    setHistoryIndex((prev) => prev + 1);
  }, [historyIndex]);

  const undo = useCallback(() => { 
    if (historyIndex > 0) setHistoryIndex((prev) => prev - 1); 
  }, [historyIndex]);

  const redo = useCallback(() => { 
    if (historyIndex < history.length - 1) setHistoryIndex((prev) => prev + 1); 
  }, [historyIndex, history.length]);

  const resetHistory = useCallback((nodes: NodeData[], edges: EdgeData[]) => {
    setHistory([{ nodes, edges }]);
    setHistoryIndex(0);
  }, []);

  const currentHistory = history[historyIndex] || { nodes: [], edges: [] };

  return {
    nodes: currentHistory.nodes,
    edges: currentHistory.edges,
    pushHistory,
    undo,
    redo,
    resetHistory,
    canUndo: historyIndex > 0,
    canRedo: historyIndex < history.length - 1
  };
}

import { RefObject, useCallback, useEffect, useRef, useState } from "react";

export function useViewport(containerRef: RefObject<HTMLDivElement | null>) {
  const [pan, setPan] = useState({ x: 100, y: 100 });
  const [scale, setScale] = useState(1);
  
  // Refs to bypass stale closures in passive event listeners
  const panRef = useRef(pan);
  const scaleRef = useRef(scale);

  useEffect(() => { panRef.current = pan; }, [pan]);
  useEffect(() => { scaleRef.current = scale; }, [scale]);

  // Native wheel event listener for high-performance trackpad zooming
  useEffect(() => {
    const container = containerRef.current;
    if (!container) return;

    const handleWheel = (e: WheelEvent) => {
      e.preventDefault();
      if (e.ctrlKey || e.metaKey) {
        const zoomFactor = e.deltaY > 0 ? 0.9 : 1.1;
        const newScale = Math.max(0.1, Math.min(scaleRef.current * zoomFactor, 5));
        const rect = container.getBoundingClientRect();
        const x = e.clientX - rect.left;
        const y = e.clientY - rect.top;
        
        setScale(newScale);
        setPan({
          x: x - (x - panRef.current.x) * (newScale / scaleRef.current),
          y: y - (y - panRef.current.y) * (newScale / scaleRef.current),
        });
      } else {
        setPan((prev) => ({ x: prev.x - e.deltaX, y: prev.y - e.deltaY }));
      }
    };
    
    container.addEventListener("wheel", handleWheel, { passive: false });
    return () => container.removeEventListener("wheel", handleWheel);
  }, [containerRef]);

  // Button-driven zoom logic
  const adjustZoomCentered = useCallback((factor: number) => {
    const rect = containerRef.current?.getBoundingClientRect();
    if (!rect) return;
    const newScale = Math.max(0.1, Math.min(scale * factor, 5));
    setPan(prevPan => ({
      x: rect.width / 2 - (rect.width / 2 - prevPan.x) * (newScale / scale),
      y: rect.height / 2 - (rect.height / 2 - prevPan.y) * (newScale / scale),
    }));
    setScale(newScale);
  }, [scale, containerRef]);

  const resetZoom = useCallback(() => {
    setScale(1);
    setPan({ x: 100, y: 100 });
  }, []);

  // Logical coordinate projection for mouse clicks
  const getLogicalCoords = useCallback((clientX: number, clientY: number) => {
    const rect = containerRef.current?.getBoundingClientRect();
    if (!rect) return { x: 0, y: 0 };
    return {
      x: (clientX - rect.left - pan.x) / scale,
      y: (clientY - rect.top - pan.y) / scale,
    };
  }, [pan, scale, containerRef]);

  return { 
    pan, 
    setPan, 
    scale, 
    setScale, 
    adjustZoomCentered, 
    resetZoom, 
    getLogicalCoords 
  };
}

"use client";

import { RefObject, useEffect, useRef, useState } from "react";

// 1. Tell TypeScript that 'google' exists on the global window object
declare global {
  interface Window {
    google: typeof google;
  }
}

export interface AddressDetails {
  streetAddress: string;
  city: string;
  stateProvince: string;
  postalCode: string;
  country: string;
  countryCode: string;
}

export function useGooglePlacesAutocomplete(
  inputRef: RefObject<HTMLInputElement | null>,
  countryCode: string | undefined | null,
  onAddressSelect: (details: AddressDetails) => void
) {
  // 2. Fix "cascading renders" warning by initializing state safely based on the window object
  const [isLoaded, setIsLoaded] = useState<boolean>(() => {
    if (typeof window !== "undefined" && window.google?.maps?.places) {
      return true;
    }
    return false;
  });

  const autocompleteRef = useRef<google.maps.places.Autocomplete | null>(null);

  const onAddressSelectRef = useRef(onAddressSelect);
  useEffect(() => {
    onAddressSelectRef.current = onAddressSelect;
  }, [onAddressSelect]);

  // Load the script only if not already loaded
  useEffect(() => {
    if (isLoaded) return;

    const scriptId = "google-maps-script";
    let script = document.getElementById(scriptId) as HTMLScriptElement;

    if (!script) {
      script = document.createElement("script");
      script.id = scriptId;
      script.src = `https://maps.googleapis.com/maps/api/js?key=${process.env.NEXT_PUBLIC_GOOGLE_MAPS_API_KEY}&libraries=places`;
      script.async = true;
      script.defer = true;
      script.onload = () => setIsLoaded(true);
      document.head.appendChild(script);
    } else {
      const handleLoad = () => setIsLoaded(true);
      script.addEventListener("load", handleLoad);
      return () => script.removeEventListener("load", handleLoad);
    }
  }, [isLoaded]);

  // 3. Store the country code in a ref to use during initial mount without triggering the useEffect dependency warning
  const countryCodeRef = useRef(countryCode);
  useEffect(() => {
    countryCodeRef.current = countryCode;
  }, [countryCode]);

  // Initialize Autocomplete
  useEffect(() => {
    if (!isLoaded || !inputRef.current) return;

    // 4. Remove 'any' and use standard Google Maps types
    const options: google.maps.places.AutocompleteOptions = {
      fields: ["address_components", "formatted_address", "name"],
      types: ["address"],
    };

    const initialCountryCode = countryCodeRef.current;
    if (initialCountryCode && initialCountryCode.trim().length > 0) {
      options.componentRestrictions = { country: initialCountryCode.toLowerCase() };
    }

    autocompleteRef.current = new window.google.maps.places.Autocomplete(
      inputRef.current,
      options
    );

    const listener = autocompleteRef.current.addListener("place_changed", () => {
      const place = autocompleteRef.current?.getPlace();
      if (!place || !place.address_components) return;

      let streetNumber = "";
      let route = "";
      let city = "";
      let stateProvince = "";
      let postalCode = "";
      let country = "";
      let returnedCountryCode = "";

      for (const component of place.address_components) {
        const types = component.types;
        if (types.includes("street_number")) streetNumber = component.long_name;
        if (types.includes("route")) route = component.long_name;

        if (types.includes("locality")) {
          city = component.long_name;
        } else if (!city && types.includes("sublocality_level_1")) {
          city = component.long_name;
        } else if (!city && types.includes("administrative_area_level_2")) {
          city = component.long_name;
        }

        if (types.includes("administrative_area_level_1")) stateProvince = component.long_name;
        if (types.includes("postal_code")) postalCode = component.long_name;
        if (types.includes("country")) {
          country = component.long_name;
          returnedCountryCode = component.short_name;
        }
      }

      onAddressSelectRef.current({
        streetAddress: streetNumber || route ? `${streetNumber} ${route}`.trim() : (place.name || ""),
        city,
        stateProvince,
        postalCode,
        country,
        countryCode: returnedCountryCode,
      });
    });

    return () => {
      if (window.google?.maps?.event && listener) {
        window.google.maps.event.removeListener(listener);
      }
      document.querySelectorAll(".pac-container").forEach((node) => node.remove());
    };
  }, [isLoaded, inputRef]); // 'countryCode' is removed from deps because we read it safely via countryCodeRef

  // Handle dynamic country updates
  useEffect(() => {
    if (autocompleteRef.current) {
      if (countryCode && countryCode.trim().length > 0) {
        autocompleteRef.current.setComponentRestrictions({ country: countryCode.toLowerCase() });
      } else {
        // 5. Remove 'null as any' - passing null natively clears restrictions
        autocompleteRef.current.setComponentRestrictions(null);
      }
    }
  }, [countryCode]);

  return { isLoaded };
}.

import * as React from "react"

const MOBILE_BREAKPOINT = 768

export function useIsMobile() {
  const [isMobile, setIsMobile] = React.useState<boolean | undefined>(undefined)

  React.useEffect(() => {
    const mql = window.matchMedia(`(max-width: ${MOBILE_BREAKPOINT - 1}px)`)
    const onChange = () => {
      setIsMobile(window.innerWidth < MOBILE_BREAKPOINT)
    }
    mql.addEventListener("change", onChange)
    setIsMobile(window.innerWidth < MOBILE_BREAKPOINT)
    return () => mql.removeEventListener("change", onChange)
  }, [])

  return !!isMobile
}

import axios from 'axios';

export const axiosInstance = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_URL || 'http://localhost:8081',
});

// Intercept all outgoing requests and attach the Bearer token dynamically
axiosInstance.interceptors.request.use(
  async (config) => {
    let token = null;

    if (typeof window !== 'undefined') {
      // Client-Side: Extract token from document.cookie
      const match = document.cookie.match(new RegExp('(^| )echonav_token=([^;]+)'));
      if (match) token = match[2];
    } else {
      // Server-Side (Next.js 16 SSR context)
      try {
        const { cookies } = await import('next/headers');
        const cookieStore = await cookies();
        token = cookieStore.get('echonav_token')?.value;
      } catch (error) {
        // Ignore if cookies() is not available in the current server context
      }
    }

    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);.
